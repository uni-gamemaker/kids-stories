    Use it as an API 
    # 1) create a job from direct links
curl -X POST $BASE/api/jobs -H 'Content-Type: application/json' \
  -d '{"videos":["https://.../1.mp4","https://.../2.mp4"]}'      # -> {"jobId": "..."}
# 2) run it (takes seconds to a few minutes)
curl -X POST $BASE/api/jobs/JOB_ID/process
# 3) status / result
curl $BASE/api/jobs/JOB_ID      # -> {"status":"done","videoUrl":"..."}
curl -X DELETE $BASE/api/jobs/JOB_ID
